% Section heading: uppercase, accent, hairline rule beneath.
% Asymmetric spacing: generous above, tight below, so each heading binds to
% the block it introduces instead of floating between two.
\newcommand{\cvsection}[1]{%
  \vspace{3pt}%
  {\normalsize\bfseries\color{accent}%
   \addfontfeature{LetterSpace=12}\MakeUppercase{#1}}\\[2pt]%
  \textcolor{rulecol}{\hrule}%
  \vspace{3pt}%
}

  {\textcolor{accent}{\textbf{\small #2}}%
   \ifthenelse{\equal{#3}{}}{}{{\color{muted}\small\;\textperiodcentered\;#3}}%

  \ifthenelse{\equal{#4}{}}{}{\foreach \d in {#4}{{\small\d}\\[-1pt]}}%
  \vspace{2pt}%
}
